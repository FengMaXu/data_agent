

describe("evidence-plan-v2 integration", () => {
  it("freezes protocol, rejects self-reported trust, and requires Host clarification evidence", async () => {
    const assurance = new InMemoryQueryAssurance({ digestCompiler: authoritativeDigestCompiler });
    const task = await assurance.prepareTask({
      question: "Count completed orders",
      planProtocolVersion: "evidence-plan-v2",
      hypotheses: [{ id: "H1", statement: "completed maps to done", basis: "enum", impact: "population", scope: "filters", confidence: 0.9, status: "candidate" }],
    }, new AbortController().signal);
    expect(assurance.taskProtocol(task.taskId)?.protocolVersion).toBe("evidence-plan-v2");
    const spec = task.answerSpec!;
    expect(() => assurance.reviseSevenFacets(task.taskId, spec.specVersion, {
      facets: spec.facets!, facetStatus: spec.facetStatus!, hypothesisBindings: spec.hypothesisBindings!,
      hypotheses: spec.hypotheses.map((item) => ({ ...item, status: "user_confirmed" as const })),
    })).toThrow("LEGACY_TRUSTED_STATUS_REJECTED:H1:user_confirmed");
    expect(() => assurance.applyClarification(task.taskId, spec.specVersion, "Use done")).toThrow("CLARIFICATION_EVENT_REQUIRED");

    const evidence = {
      evidenceId: "EV-HOST-1", taskId: task.taskId, sourceKind: "clarification" as const,
      sourceIdentity: "clarification:1", sourceRevision: "1", locator: "answer",
      contentHash: evidenceContentHash("Use done"), authority: "user_clarification" as const,
      observedAt: "2026-01-01T00:00:00.000Z", hostEventId: "host-event-1",
    };
    const revised = assurance.applyClarificationEvidence(task.taskId, spec.specVersion, "H1", "Use done", evidence);
    expect(revised.specVersion).not.toBe(spec.specVersion);
    expect(assurance.projectedHypothesisStatus(task.taskId, revised.specVersion, "H1")).toBe("user_confirmed");
    expect(assurance.projectedHypothesisStatus(task.taskId, spec.specVersion, "H1")).toBe("candidate");
  });

  it("restores task protocol and evidence/verification relationships", async () => {
    const store = new InMemoryQueryAssuranceStateStore();
    const first = new InMemoryQueryAssurance({ digestCompiler: authoritativeDigestCompiler, stateStore: store });
    const task = await first.prepareTask({ question: "Count completed orders", planProtocolVersion: "evidence-plan-v2", hypotheses: [{ id: "H1", statement: "completed maps to done", basis: "guess", impact: "population", scope: "filters", confidence: 0.5, status: "candidate" }] }, new AbortController().signal);
    const evidence = { evidenceId: "EV-HOST-2", taskId: task.taskId, sourceKind: "clarification" as const, sourceIdentity: "clarification:2", sourceRevision: "1", locator: "answer", contentHash: evidenceContentHash("Use done"), authority: "user_clarification" as const, observedAt: "2026-01-01T00:00:00.000Z", hostEventId: "host-event-2" };
    const revised = first.applyClarificationEvidence(task.taskId, task.specVersion!, "H1", "Use done", evidence);
    const restored = new InMemoryQueryAssurance({ digestCompiler: authoritativeDigestCompiler, stateStore: store });
    expect(restored.taskProtocol(task.taskId)?.protocolVersion).toBe("evidence-plan-v2");
    expect(restored.projectedHypothesisStatus(task.taskId, revised.specVersion, "H1")).toBe("user_confirmed");
    expect(() => restored.applyClarification(task.taskId, revised.specVersion, "change")).toThrow("CLARIFICATION_EVENT_REQUIRED");
  });
});
